func main(null) -> int
{
    const limit : int = 10;
    var n : int = 0;

    {
        var i : int = 0;

        while (i < limit)
        {
            if (i != 5)
            {
                n += i;
            }
            i++;
        }
        n = n + 5;
    }
    return n;
}
