func fibonacci(n : int) -> int
{
    if (n < 2)
    {
        return n;
    }
    else
    {
        return fibonacci(n - 2) + fibonacci(n - 1);
    }
}

func main(null) -> int
{
    var start : float = clock();
    var n : int = fibonacci(35);
    print(clock() - start);
    print(n);
    return 0;
}
